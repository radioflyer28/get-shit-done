Create `get-shit-done/bin/sbom_generate.sh`:

```bash
#!/usr/bin/env bash
set -euo pipefail

TARGET="${1:-.}"
ABS_TARGET="$(cd "$TARGET" && pwd)"
SBOM_FILE="${ABS_TARGET}/SBOM.json"

# Tool detection and invocation
if command -v syft &>/dev/null; then
  echo "Generating SBOM with syft..."
  syft "${ABS_TARGET}" -o cyclonedx-json="${SBOM_FILE}"
  echo "✓ SBOM written to SBOM.json (CycloneDX JSON via syft)"
elif command -v cyclonedx-cli &>/dev/null; then
  echo "Generating SBOM with cyclonedx-cli..."
  cyclonedx-cli analyze --directory "${ABS_TARGET}" \
    --output-format json --output-file "${SBOM_FILE}"
  echo "✓ SBOM written to SBOM.json (CycloneDX JSON via cyclonedx-cli)"
else
  echo "⚠ SBOM generation requires syft or cyclonedx-cli"
  echo ""
  echo "Install syft:         curl -sSfL https://raw.githubusercontent.com/anchore/syft/main/install.sh | sh -s -- -b /usr/local/bin"
  echo "Install cyclonedx-cli: https://github.com/CycloneDX/cyclonedx-cli/releases"
  exit 1
fi

echo "SBOM packages: $(python3 -c "import json,sys; d=json.load(open('${SBOM_FILE}')); print(len(d.get('components',[])), 'components')" 2>/dev/null || echo 'unknown')"
```

The script must not execute any input from TARGET directory (only passes the path as argument to syft/cyclonedx-cli).
